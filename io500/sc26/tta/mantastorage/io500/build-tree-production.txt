+/*
+ * Look up an entry without the size/epoch part of dfs_stat(): one fetch_entry,
+ * the object open is local. Returns the type in mode and an open object that
+ * must be released with pfind_dfs_release(). O_NOFOLLOW keeps the lstat
+ * semantics of pfind_dfs_lstat() for symlinks.
+ */
+int pfind_dfs_lookup(const char* path, mode_t* mode, dfs_obj_t** obj)
+{
+    int rc;
+    dfs_obj_t *parent = NULL;
+    char *name = NULL, *dir_name = NULL;
+
+    rc = parse_filename(path, &name, &dir_name);
+    if (rc) {
+	    fprintf(stderr, "parse_filename %s failed (%d)\n", path, rc);
+	    if (rc < 0)
+		    rc = -rc;
+	    goto out;
+    }
+
+    assert(dir_name);
+
+    parent = lookup_insert_dir(dir_name, NULL);
+    if (parent == NULL) {
+	    fprintf(stderr, "dfs_lookup %s failed \n", dir_name);
+	    rc = ENOENT;
+	    goto out;
+    }
+
+    rc = dfs_lookup_rel(pfind_dfs, parent, name, O_RDONLY | O_NOFOLLOW, obj, mode, NULL);
+    if (rc)
+	    fprintf(stderr, "dfs_lookup_rel %s failed (%d)\n", name, rc);
+
+out:
+    free(name);
+    free(dir_name);
+    return rc;
+}
